ENG_section_war_plans_category = {

    ENG_section_atlantis_war_plans = {

        icon = GFX_decision_hol_draw_up_staff_plans

        days_remove = 60

        allowed = {
            original_tag = ENG
        }

        target_root_trigger = {
            has_completed_focus = ENG_section_atlantis
        }

        target_trigger = {
            FROM = {
                has_war_with = ROOT

                any_controlled_state = {
					is_on_continent = north_america
                }
            }
        }

        visible = {
            has_war_with = FROM

            FROM = {
                NOT = {
                    has_country_flag = ENG_war_plan_target_cooldown
                }
            }

            OR = {
                NOT = {
                    has_country_flag = ENG_war_plan_active
                }
                FROM = {
                    has_country_flag = ENG_war_plan_target_active
                }
            }
        }

        available = {
            NOT = {
                has_country_flag = ENG_war_plan_active
            }
        }

        custom_cost_trigger = {
            command_power > 24
            has_army_experience > 14
        }

        custom_cost_text = decision_cost_CP_25_ArmXP_15

        remove_effect = {
            FROM = {
                set_country_flag = {
                    flag = ENG_war_plan_target_cooldown
                    days = 60
                    value = 1
                }
            }
        }

        complete_effect = {
            hidden_effect = {
                add_command_power = -25
                army_experience = -15
            }

            set_country_flag = {
                flag = ENG_war_plan_active
                days = 60
                value = 1
            }

            FROM = {
                set_country_flag = {
                    flag = ENG_war_plan_target_active
                    days = 60
                    value = 1
                }

                every_controlled_state = {
                    limit = {
						is_on_continent = north_america
                    }

                    add_dynamic_modifier = {
                        modifier = ENG_north_america_war_plan_modifier
                        scope = ROOT
                        days = 60
                    }
                }
            }
        }
    }
