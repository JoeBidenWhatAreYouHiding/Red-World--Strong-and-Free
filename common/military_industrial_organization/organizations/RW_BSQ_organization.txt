sapa = {
	include = generic_mobile_tank_organization	
	icon = GFX_idea_sapa_company
	
	allowed = {	
		original_tag = BSQ
	}
	
	available = {
		# When in a Foreign MIO, countries need to be at peace with original country
		IF = {
			limit = {
				FROM = { NOT = { original_tag = BSQ } }
			}
			FROM = { NOT = { has_war_with = BSQ } }
		}
	}
}
