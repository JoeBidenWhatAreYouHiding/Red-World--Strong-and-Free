#Tanks
enasa_tank = {
	include = generic_mobile_tank_organization	
	icon = GFX_idea_enasa_company
	
	allowed = {	
		original_tag = CAT
	}
	
	available = {
		# When in a Foreign MIO, countries need to be at peace with original country
		IF = {
			limit = {
				FROM = { NOT = { original_tag = CAT } }
			}
			FROM = { NOT = { has_war_with = CAT } }
		}
	}
}
