AMR_is_soc_unifier = {
	OR = {
		AND = {
			has_government = revolutionary_socialist
			NOT = { tag = AMC }
		}
		AND = {
			has_government = moderate_socialist
			has_country_leader = {
				name = "Winona LaDuke"
				ruling_only = yes
			}
		}
	}
}
