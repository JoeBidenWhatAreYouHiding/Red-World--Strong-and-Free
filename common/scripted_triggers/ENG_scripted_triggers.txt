ENG_can_join_revsoc_faction = {
	custom_trigger_tooltip = {
		tooltip = ENG_can_join_revsoc_faction_tt
		is_in_faction = no
		has_government = revolutionary_socialist
		any_country = {
			NOT = {
				tag = UAR #theirs is a local faction
			}
			has_government = revolutionary_socialist
			is_faction_leader = yes
		}
	}
}

ENG_can_join_warsaw_pact = {
	custom_trigger_tooltip = {
		tooltip = ENG_can_join_warsaw_pact_tt
		is_in_faction = no
		has_government = communist
		OR = {
			has_completed_focus = ENG_workers_revolution
			has_completed_focus = ENG_total_revolution #TODO change based on what subideology rules the WP
		}
		OR = {
			AND = {
				country_exists = SOV
				SOV = {
					is_faction_leader = yes
					has_government = communist
				}
			}
			AND = { #TODO factor in RevSoc Germany
				country_exists = GER
				GER = {
					is_faction_leader = yes
					has_government = communist
				}
			}
		}
	}
}
