		remove_ideas = TEX_weaponry_seizures2
		remove_ideas = TEX_weaponry_seizures3
		remove_ideas = TEX_houston_vindicated
		remove_ideas = TEX_crusade_against_yankeedom
		remove_ideas = TEX_intranet_links
		remove_ideas = TEX_intranet_links2
		remove_ideas = TEX_ballad_of_texas_red
		remove_ideas = TEX_counter_attack_focus
		remove_ideas = TEX_propaganda_effort
		remove_ideas = TEX_american_proxy_war
		remove_ideas = TEX_Focus_on_Offense4
		remove_ideas = TEX_Focus_on_Defense4
		remove_ideas = TEX_Remeber_March_Seventeen
		if = {
			limit = { 
				NOT = { has_country_flag = TEX_keep_spirit_tt }
			}
			remove_ideas = TEX_ballad_of_texas_red
		}
	}
	custom_effect_tooltip = TEX_remove_wartime_spirits_idea
}
	
	
TEX_decrease_american_legacy_in_state = {
	if = {
		limit = { has_dynamic_modifier = { modifier = TEX_American_Legacy } }
		hidden_effect = {
			remove_dynamic_modifier = { modifier = TEX_American_Legacy }
			add_dynamic_modifier = { modifier = TEX_American_Legacy_2 }
		}
		custom_effect_tooltip = CAN_resistance_effects_reduced
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = TEX_American_Legacy_2 } }
		hidden_effect = {
			remove_dynamic_modifier = { modifier = TEX_American_Legacy_2 }
			add_dynamic_modifier = { modifier = TEX_American_Legacy_3 }
		}
		custom_effect_tooltip = CAN_resistance_effects_reduced
	}
	else_if = {
		hidden_effect = {
			limit = { has_dynamic_modifier = { modifier = TEX_American_Legacy_3 } }
			remove_dynamic_modifier = { modifier = TEX_American_Legacy_3 }
		}
		custom_effect_tooltip = CAN_resistance_effects_removed
	}
}
